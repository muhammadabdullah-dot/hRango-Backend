## 6. Leave [lv]

```
leave_type                            Settings -> Leave Types ("Add Leave Types")
  id                PK
  company_id        FK company
  title, short_code
  allowance                           No of Leaves Allowed, per tenure
  is_active, is_probation_type, auto_allotment, auto_balance_on_joining
  is_composite, is_short_leave, show_in_payslip, is_deductible
  is_sandwich, sandwich_across_types, is_earned
  allow_back_date, back_date_limit_days
  has_future_limit, future_limit_days
  attachment_required
  attachment_rule                     refuse, note (Do When Attachment is Required)
  allow_encashment, encashment_min, encashment_max
  carry_forward, pro_rated
  gender {NN}                         any, male, female
  marital_status {NN}                 any, single, married
  allowed_in_probation
  min_experience_months, max_applications, gap_between_applications_months
  tenure {NVARCHAR(30)} {NN}          yearly, monthly, employment
  balance_basis {NN}                  flat, experience
  accumulative_count {INT}            tenures unused days may pile up over; NULL = no cap
  UNIQUE (company_id, title)

leave_type_experience_slab            Balance According to = Experience
  id PK, leave_type_id FK leave_type, min_months {NN}, max_months, balance {NN}

leave_type_short_slab                 with Short Leave
  id PK, leave_type_id FK leave_type, name {NN}, balance, value_type {NN}, slab_value {DECIMAL(9,2)}

leave_policy                          Settings -> Leave Policies
  id                PK
  company_id        FK company
  title
  policy_kind {NN}                    distribution, half_short
  policy_level {NN}                   company (everyone), employee (the enrolled)
  allow_half_day, allow_short_leave   half_short policies
  is_active
  UNIQUE (company_id, title)

leave_policy_type                     the types a distribution policy gives, with its own allowance
  leave_policy_id FK leave_policy, leave_type_id FK leave_type, allowance     *──*

leave_policy_employee                 enrolment in an employee-level policy
  leave_policy_id FK leave_policy, employee_id FK employee                  *──*

leave_ledger                          every change to a balance; balance = sum
  id                PK
  employee_id       FK employee
  leave_type_id {NN}  FK leave_type
  fiscal_year                         leave year (year the company fiscal year starts in)
  entry_date {NN}
  days                                + credit, − debit
  reason                              opening, allotment, earned, carry_forward, adjustment, overtime, availed, encashed, penalty, restored
  remarks
  request_id        FK request        optional; availed and restored postings
  leave_encashment_id FK leave_encashment   optional; encashed postings
  warning_id        FK warning        optional; penalty postings

leave_encashment                      Payroll -> Leave Encashment (one run for a type and year)
  id                PK
  company_id        FK company
  leave_type_id {NN}  FK leave_type
  fiscal_year {NN}
  encashment_date {NN}
  applies_to_all
  basis {NN}                          overtime, earned_salary, per_hour_rate, overnight
  amount_per_leave {NN}
  mode {NN}                           payroll, manual

leave_encashment_employee
  id PK, leave_encashment_id FK leave_encashment, employee_id {NN} FK employee, days {NN}, amount {NN}, payslip_id FK payslip
  UNIQUE (leave_encashment_id, employee_id)
```

```
leave_type 1──* leave_type_experience_slab / leave_type_short_slab
leave_policy *──* leave_type / employee
employee 1──* leave_ledger *──1 leave_type
leave_encashment 1──* leave_encashment_employee
leave_ledger *──1 request / leave_encashment / warning   (what posted it)
```
