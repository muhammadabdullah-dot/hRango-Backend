## 2. Employee [hr]

```
employee
  id                PK
  company_id        FK company
  code                                prefix from branch/location + number
  first_name, last_name, parent_name
  gender, marital_status, religion, nationality
  date_of_birth, cnic                 cnic is sensitive (employees.identity)
  email, phone_1, phone_2
  home_address {NVARCHAR(500)}        Personal tab / Benefits "Home Address"; changed by HR or an approved Update Information request
  home_city_id      FK city          optional
  photo_file_id     FK stored_file    profile photo (an image upload); responses give photoUrl, a signed link
  branch_id         FK branch
  location_id       FK location
  area_id           FK area
  department_id     FK department
  division_id       FK division
  cost_center_id    FK cost_center
  designation_id    FK designation
  job_title_id      FK job_title
  grade_id          FK grade
  job_level_id      FK job_level
  employee_group_id FK employee_group
  employment_type_id FK employment_type
  line_manager_id   FK employee       no cycles: nobody manages their own manager
  final_authority_id FK employee      used by approval flows
  default_time_slot_id FK time_slot
  status                              active, probation, notice, left, inactive (switched off by HR)
  salary_frequency                    monthly, daily, hourly

employment_period                     one row per joining; rehire adds a row
  id                PK
  employee_id       FK employee
  joining_date {NN}
  confirmation_date, probation_end_date
  end_date
  end_of_employment_id FK end_of_employment

employee_history                      what changed and when: joining, transfer, promotion, demotion, edit
  id                PK
  employee_id       FK employee
  change_type {NN}                    joining, transfer, promotion, demotion, edit
  effective_date {NN}
  location_id       FK location
  branch_id         FK branch
  department_id     FK department
  designation_id    FK designation
  job_title_id      FK job_title
  grade_id          FK grade
  job_level_id      FK job_level
  line_manager_id   FK employee
  final_authority_id FK employee
  source_request_id FK request
  previous_values {NVARCHAR(MAX)}     JSON of the placement before this change; a cancelled request puts it back
  is_reverted                         set when the request behind it is cancelled after approval

employee_salary                       salary over time; the Salary Revision screen lists this
  id                PK
  employee_id       FK employee
  effective_date
  amount
  previous_amount
  change_type                         flat, percent
  change_value
  source                              joining, revision, promotion, import
  source_request_id FK request
  remarks

employee_pay_item                     fixed allowances / deductions per person
  id PK, employee_id FK employee, pay_item_id FK pay_item, amount, effective_date

employee_bank_account                 sensitive (employees.bank_accounts)
  id                PK
  employee_id       FK employee
  account_holder, account_number {NN}, bank_name {NN}
  branch_name, branch_code, tax_payer_id
  is_primary

employee_document                     Documents tab (SRS EMP-3, APP Documents): CNIC copy, degree, letters
  id                PK
  employee_id       FK employee
  document_type {NN}                  cnic, passport, driving_licence (identity: employees.identity too);
                                      offer_letter, promotion_letter, appraisal_letter, salary_certificate (pay: employees.salary too);
                                      experience_letter, degree, certificate, medical, other
  title {NULL}
  document_number {NVARCHAR(100)}
  issue_date, expiry_date
  file_id           FK stored_file {NN}
  remarks
  expiry_notified_at                  set when the employee was told it expires soon (hr.usp_employee_document_expiry_notify)

employee_experience                   Professional experience: Company Name, Job Title, From / To (to empty = present)
  id PK, employee_id FK employee, company_name {NN}, role_title {NN}, from_date {NN}, to_date

employee_agreement
  id PK, employee_id FK employee, file_id FK stored_file, accepted_at

employee_emergency_contact            Emergency Contact # on the Personal tab is the primary contact's phone
  id                PK
  employee_id       FK employee
  name {NN}
  relation {NN}
  phone {NN}
  alt_phone
  address
  is_primary

employee_dependent                    Dependents tab: First Name, Last Name, Relation, Occupation
  id                PK
  employee_id       FK employee
  first_name
  last_name {NULL}
  relation {NN}                       spouse, son, daughter, father, mother, brother, sister, other
  occupation
  date_of_birth

employee_skill                        Skills tab: Skill Title, Skill Level
  id                PK
  employee_id       FK employee
  skill_title {NN}
  skill_level {NN}                    beginner, intermediate, advanced, expert
  UNIQUE (employee_id, skill_title)

employee_education                    Add Academic: Academic Title, Institute, From / To, In Process, document
  id                PK
  employee_id       FK employee
  title
  institute {NN}
  from_date, to_date
  is_in_process
  file_id           FK stored_file

employee_contract                     Add Contract: Contract Type, Duration in Months (or Custom), From / To, attachment.
                                      Sensitive (employees.contracts): a contract states pay
  id                PK
  employee_id       FK employee
  contract_type {NN}                  permanent, temporary, contractual, probation, internship, consultant
  duration_months                     empty = custom dates
  from_date {NN}
  to_date
  file_id           FK stored_file
  remarks

employee_change_request               "Update Information": an employee asks, HR approves; approval writes hr.employee
  id                PK
  company_id        FK company
  employee_id       FK employee {NN}
  status                              pending, approved, rejected, cancelled
  changes {NVARCHAR(MAX)} {NN}        JSON of the asked values (snake_case keys): date_of_birth, marital_status,
                                      phone_1, phone_2, home_address, home_city_id
  previous_values {NVARCHAR(MAX)}     JSON of the same fields when asked
  remarks
  decided_by_user_id FK user_account
  decided_at
  decision_remarks {NVARCHAR(MAX)}
```

```
employee 1──* employment_period / employee_history / employee_salary
employee 1──* employee_bank_account / employee_document / employee_experience
employee 1──* employee_emergency_contact / employee_dependent / employee_skill / employee_education
employee 1──* employee_contract / employee_change_request
employee 1──* employee (line manager → team)
```
